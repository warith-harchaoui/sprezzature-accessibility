# Requirements for alt_from_ollama.py. The floors follow pyproject.toml,
# which is what `pip install sprezzature-accessibility` installs.
#
# Install:
#   pip install -r scripts/requirements-alt-text.txt
#
# Used for:
#   - click                : the command-line parser.
#   - langdetect           : detect language from --in DOC / --context text
#                            so the output language matches the page without
#                            relying on shell locale.
#   - Pillow               : downscales large images before sending them to
#                            the model (`--resize`). A base dependency in
#                            pyproject.toml since 2026-11-04.
#   - best-engine-ai-helper: routes the actual LLM/VLM call (Ollama by
#                            default; see SPREZZATURE_LLM_* env vars).
#   - wallet-helper        : memoizes those calls — content-addressed, bounded,
#                            and single-flight across processes, which is the
#                            one of the four the cache written here could not
#                            do: three runs started together on one cold image
#                            called the model three times.

click>=8.1
langdetect>=1.0.9
Pillow>=10.0.0
best-engine-ai-helper>=0.4.0
wallet-helper>=1.3.2,<2
